\section*{Metrics}
\begin{align}
\ell_{n,r} &= \underbrace{\frac{\Delta_{n,r}}{10^{3}}}_{\mu\mathrm{s}\,\to\,\mathrm{ms}},
\qquad \Delta_{n,r}=T^{\mathrm{stop}}_{n,r}-T^{\mathrm{start}}_{n,r}
\label{eq:latency}\\
\theta_{n,r} &= \underbrace{\frac{n}{\ell_{n,r}/10^{3}}}_{\text{respondents per second}}
\label{eq:throughput}\\
\bar\ell_n &= \frac{1}{R_n}\sum_{r=1}^{R_n}\ell_{n,r},
\qquad
s_n=\sqrt{\frac{1}{R_n-1}\sum_{r=1}^{R_n}\bigl(\ell_{n,r}-\bar\ell_n\bigr)^{2}}
\label{eq:meansd}
\end{align}

With order statistics $\ell_{(1)}\le\dots\le\ell_{(R)}$ and $h_q=(R-1)\,q/100$,
\begin{equation}
\hat p_q=\ell_{(\lfloor h_q\rfloor+1)}+\underbrace{\bigl(h_q-\lfloor h_q\rfloor\bigr)}_{\text{interpolation weight}}
\bigl(\ell_{(\lfloor h_q\rfloor+2)}-\ell_{(\lfloor h_q\rfloor+1)}\bigr),
\quad q\in\{50,95,99\}.
\label{eq:percentile}
\end{equation}

The 95\% percentile-bootstrap confidence interval of a statistic $T$ is
\begin{equation}
\mathrm{CI}_{95}(T)=\Bigl[\,Q_{0.025}\bigl(\{T^{*(b)}\}_{b=1}^{B}\bigr),\;
Q_{0.975}\bigl(\{T^{*(b)}\}_{b=1}^{B}\bigr)\Bigr],
\quad T^{*(b)}=T\bigl(\boldsymbol{\ell}_n[\mathbf{J}_{b,\cdot}]\bigr),
\label{eq:bootstrap}
\end{equation}
where row $\mathbf{J}_{b,\cdot}$ draws $R_n$ indices with replacement (seed $42$).

\begin{align}
\varepsilon &= \frac{1}{R_{\mathrm{all}}}\sum_{r=1}^{R_{\mathrm{all}}}
\mathbf{1}\bigl[\mathrm{status}_r=\text{error}\bigr]
\label{eq:error}\\
\ell \text{ is flagged} &\iff \ell<Q_1-1.5\,\mathrm{IQR}\ \ \text{or}\ \ \ell>Q_3+1.5\,\mathrm{IQR}
\label{eq:iqr}\\
S(n)&=\frac{\tilde\theta_{n}}{\tilde\theta_{n_0}},\qquad
E(n)=\frac{S(n)}{n/n_0},\qquad n_0=35
\label{eq:speedup}
\end{align}
where $\tilde\theta_n$ is the median throughput at workload $n$.
